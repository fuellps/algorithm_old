### 二维几何

##### 模板

```c++
const double pi = acos(-1.0);
const double eps = 1e-8;
int sgn(double x) {
	if (fabs(x) < eps) return 0;
	return x < 0 ? -1 : 1;
}
int dcmp(double x, double y) {
	if (fabs(x - y) < eps) return 0;
	return x < y ? -1 : 1;
}

//.........   点(向量)，点积，叉积    ....................
struct Point {
	double x, y;
	Point() {}
	Point(double _x, double _y): x(_x), y(_y) {}
    
	Point operator+(Point b) {return {x + b.x, y + b.y};}
	Point operator-(Point b) {return {x - b.x, y - b.y};}
	double operator^(const Point& b) { //叉积
		return x * b.x + y * b.y;
	}
	double operator*(const Point& b) { //点积
		return x * b.y - y * b.x;
	}
};

double dist(const Point& p1, const Point& p2) { //返回两点之间的距离
	return sqrt((p1.x - p2.x) * (p1.x - p2.x) + (p1.y - p2.y) * (p1.y - p2.y));
}
double dist2(const Point& p1, const Point& p2) { //返回两点之间的距离平方
	return (p1.x - p2.x) * (p1.x - p2.x) + (p1.y - p2.y) * (p1.y - p2.y);
}

double len( Point& a) { //返回向量的模长
	return sqrt(a ^ a);
}
double len2( Point& a) { //返回向量的模长的平方
	return a ^ a;
}
double angle( Point& a,  Point& b) { //返回两个向量的夹角
	return acos((a ^ b) / len(a) / len(b));
}
Point rotate(Point& a, double deg) { //向量a旋转deg度
	return {a.x * cos(deg) - a.y * sin(deg), a.x * sin(deg) + a.y * cos(deg)};
}


//.........   线,点线关系   ....................
struct Line {
	Point p1, p2;
	Line() {}
	Line(Point p1, Point p2): p1(p1), p2(p2) {}
	Line(Point p, double angle) { //根据一个点和倾斜角确定直线,0 <= angle < pi
		p1 = p;
		if (sgn(angle - pi / 2) == 0) {
			p2 = p1 + Point(0, 1);
		} else {
			p2 = p1 + Point(1, tan(angle));
		}
	}
};

 
double dis_point_line(Point p, Line L) { //计算点P到直线L的距离
	return fabs((L.p2 - L.p1) * (p - L.p1)) / dist(L.p1, L.p2);
}



//.........   圆   ....................
struct Circle {
	Point c;
    double r;
    Circle(){}
    Circle(Point c,double r): c(c),r(r){}
    Circle(double x,double y,double _r){
        c=Point(x,y);
        r = _r;
    }
};


```
