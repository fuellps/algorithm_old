```c++
mt19937 rng(chrono::steady_clock::now().time_since_epoch().count());
int randint(int l, int r) {
    return uniform_int_distribution<int>(l, r)(rng);
}

void quickly_sort(int l, int r) {
    if (l >= r) return;
    swap(a[l], a[randint(l, r)]);
    int lx = l, rx = r, i = l + 1;
    while (i <= rx) {
        if (a[i] < a[lx]) {//扩充[l,lx)区域
            swap(a[i++], a[lx++]);
        } else if (a[i] == a[lx]) {//k
            i++;
        } else {//扩充（rx,r]区域
            swap(a[i], a[rx]);
            rx--;
        }
    }
    quickly_sort(l, lx - 1);
    quickly_sort(rx + 1, r);
}

void solve(){
    //快速排序
    quickly_sort(0,n-1);//n表示数组长度，下标从0开始.
}
```

应用:O(n)找到数字中第k大/第k小的数.

可以将第k大,第k小映射到数组对应的下标,这样查找就只需要根据lt,gt,比较即可找到第k大或第k小的数

```c++
#include <bits/stdc++.h>
using namespace std;
const int N = 100010;
int nums[N];
//每次只需根据边界递归一半的区间
int quickSort(int l,int r,int k){
    if(l == r) return nums[l];
    swap(nums[l],nums[l + int(rand())%(r-l+1)]);
    int lt = l,i = l+1,gt = r,pivot = nums[l];
    // [l,lt) < pivot; [lt,i-1] = pivot (gt,r] > pivot 
    while(i <= gt){
        if(nums[i] < pivot){
            swap(nums[i++],nums[lt++]);
        }else if(nums[i] > pivot){
            swap(nums[i],nums[gt--]);
        }else{
            i++;
        }
    }
    if(lt <= k && k <= gt){//在等于pivot的区间内
        return pivot;
    }else if(k < lt){//在小于pivot的区间
        return quickSort(l,lt-1,k);
    }else{
        return quickSort(gt+1,r,k);//在大于pivot的区间内
    }
     
}

int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    srand(time(0));
    int n,k;cin>>n>>k;
    for(int i = 0;i < n; i++) cin>> nums[i];
    cout << quickSort(0,n-1,k-1);//第k小的数对应排序后下标为k-1
    return 0;
}
```
