###  LSD（Least Significant Digit first）基数排序

基数排序(Radix Sort)基于**多关键字**的非比较排序.核心是比较**第i个关键字以及第i个关键字之后的比较结果**。

通过计数排序和相关辅助数组实现原地排序。

- **sa数组：表示排序后第i位对应的是原数组第i位**
- **rank数组：表示原数组第i位对应的是排序后第i位**

### 模板

```c++
/*
测试用例：
8
170 45 75 90 802 24 2 66

/*
简单理解:\
sa<-排序后第i个位置是在原数组的第几个位置
rk<-第i位对应的排第几位.
*/
void counting_sort() {
	memset(c,0,sizeof c);
	cout <<"sa init: ";
	for(int i = 1;i <= n; i++) cout << sa[i] <<" \n"[i==n];
	for (int i = 1; i <= n; i++) c[v[i]]++;
	for (int i = 1; i < 10; i++) c[i] += c[i - 1]; //求前缀和
	for(int i = n;i;i--){
		//sa[i]表示排序后第i个位置是在原数组的第几个位置
		rk[sa[i]] = c[v[sa[i]]]--;
	}
	cout <<"rank: ";
	for(int i = 1;i <= n; i++) cout << rk[i] <<" \n"[i==n];
		//rk[i]第i位对应排序后第几位
	for(int i = 1;i <= n; i++){
		sa[rk[i]] = i;
	}	
	cout <<"sa after counting_sort: ";
	for(int i = 1;i <= n; i++) cout << sa[i] <<" \n"[i==n];
}
void radio_sort() {
	for (int i = 1; i <= n; i++) sa[i] = i;
	int x = 1;
	for (int i = 0; i < 3; i++,x*=10) {
		for (int j = 1; j <= n ; j++) v[j] = a[j] / x % 10;
		counting_sort();
	}
	for(int i = 1;i <= n; i++) cout << a[sa[i]] <<" \n"[i==n];
}
void solve() {
	cin >> n;
	for (int i = 1; i <= n; i++) cin >> a[i];
	radio_sort();
	
}
```
